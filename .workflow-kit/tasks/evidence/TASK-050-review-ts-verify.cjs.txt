/** 收集顶层语句：{name, start, end, text, exported} */
function topLevel(file, src) {
  const items = [];
  for (const st of file.statements) {
    let name = null;
    if (st.name && ts.isIdentifier(st.name)) name = st.name.text;
    else if (ts.isVariableStatement(st) && st.declarationList.declarations.length) {
      const d = st.declarationList.declarations[0];
      if (ts.isIdentifier(d.name)) name = d.name.text;
    }
    const mods = ts.canHaveModifiers(st) ? ts.getModifiers(st) : undefined;
    const exported = !!(mods && mods.some((m) => m.kind === ts.SyntaxKind.ExportKeyword));
    items.push({ name, start: st.getStart(file), end: st.getEnd(), text: st.getText(file), exported, kind: ts.SyntaxKind[st.kind] });
  }
  return items;
}

const origItems = topLevel(origFile, originalSrc);
const cur = new Map();
for (const rel of TARGETS) {

  const src = fs.readFileSync(p, 'utf8');
  const f = ts.createSourceFile(path.basename(rel), src, ts.ScriptTarget.Latest, true,
    rel.endsWith('.tsx') ? ts.ScriptKind.TSX : ts.ScriptKind.TS);
  cur.set(rel, { src, items: topLevel(f, src) });
}

const fold = (s) => s.replace(/\s+/g, '');
const indentInsensitive = (s) => s.split(/\r?\n/).map((l) => l.trim()).filter((l) => l).join('\n');
const stripExport = (s) => s.replace(/^export\s+(?=(?:async\s+)?(?:function|const|let|var|class|interface|type|enum|declare)\b)/, '');

function findDecl(rel, name) {
  const e = cur.get(rel);
  if (!e) return null;
  const hit = e.items.filter((it) => it.name === name);
  if (hit.length !== 1) return null;
  return hit[0];
}

let eqBad = 0, exactAll = 0, dupProblem = 0;
out('  unit                      | 原文行 | 现状行 | 所属文件                                | foldEq lineEq charEq');

  const o = origItems.filter((it) => it.name === name);
  if (o.length !== 1) { out(`!! 原文中 ${name} 顶层声明数 = ${o.length}`); eqBad++; continue; }
  // 定位现状
  const locs = [];
  for (const [rel, e] of cur) for (const it of e.items) if (it.name === name) locs.push({ rel, it });
  if (locs.length !== 1) { out(`!! 现状中 ${name} 出现次数 = ${locs.length} (${locs.map((x) => x.rel)})`); dupProblem++; continue; }
  const ob = o[0].text, fb = locs[0].it.text;
  const obS = stripExport(ob), fbS = stripExport(fb);
  const f1 = fold(obS) === fold(fbS);
  const f2 = indentInsensitive(obS).split('\n').join('\n') === indentInsensitive(fbS).split('\n').join('\n');
  const f3 = obS.replace(/\r\n/g, '\n') === fbS.replace(/\r\n/g, '\n');
  if (f3) exactAll++;
  if (!f1 || !f2 || !f3) eqBad++;
  out(`  ${name.padEnd(24)} | ${String(ob.split('\n').length).padStart(6)} | ${String(fb.split('\n').length).padStart(6)} | ${locs[0].rel.padEnd(39)} | ${f1} ${f2} ${f3}`);
}
out('');

out(`  ⇒ 逐字等价(去空白规范化)=${eqBad === 0 ? '全部一致' : '存在不一致'}；完全字符级相同(仅去掉 export 前缀)=${exactAll}/14`);
